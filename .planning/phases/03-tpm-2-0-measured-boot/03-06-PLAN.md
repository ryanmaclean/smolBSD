    # SPDX-License-Identifier: Apache-2.0
    # tpm-vm-test.yml — TPM 2.0 T1-T6 acceptance suite on pop4090 QEMU+swtpm.
    # Runs against the smolBSD amd64 image built with 'device tpm' + tpm2-tools.
    # Image path: /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2
    # Runner: pop4090-amd64 (self-hosted, linux, amd64, kvm)
    #
    # Phase-3 acceptance gates:
    #   T1: swtpm socket appears within 3s
    #   T2: /dev/tpm0 present in guest
    #   T3: tpmctl -G returns TPM 2.0 manufacturer info
    #   T4: PCR 0 readable, non-zero
    #   T5: seal/unseal round-trip with PCR 0+7 policy
    #   T6: PCR changes after tpm2_pcrextend

    name: TPM VM Test

    on:
      push:
      workflow_dispatch:

    env:
      SMOLBSD_IMAGE: /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2
      SWTPM_STATE:   /var/run/smolbsd-tpm
      SSH_PORT:      2241
      SSH_PASS:      smolbsd

    jobs:
      tpm-t1-t6:
        name: TPM T1-T6 acceptance suite
        runs-on: [self-hosted, linux, amd64, kvm]

        steps:
          - uses: actions/checkout@v4

          - name: Install Nushell
            run: |
              NU_VERSION="0.112.2"
              ARCH=$(uname -m | sed 's/arm64/aarch64/')
              case "$(uname -s)" in
                Linux)  TRIPLE="${ARCH}-unknown-linux-musl" ;;
                Darwin) TRIPLE="${ARCH}-apple-darwin" ;;
              esac
              ARCHIVE="nu-${NU_VERSION}-${TRIPLE}.tar.gz"
              mkdir -p "$HOME/.local/bin"
              curl -sSL "https://github.com/nushell/nushell/releases/download/${NU_VERSION}/${ARCHIVE}" \
                | tar -xz -C "$HOME/.local/bin" --strip-components=1 "nu-${NU_VERSION}-${TRIPLE}/nu"
              echo "$HOME/.local/bin" >> "$GITHUB_PATH"

          - name: Show Nushell version
            run: nu --version

          - name: Verify smolBSD image exists
            run: |
              test -f "$SMOLBSD_IMAGE" || {
                echo "ERROR: smolBSD image not found at $SMOLBSD_IMAGE"
                echo "Run build-image.yml workflow first to build the image."
                exit 1
              }
              echo "Image found: $(ls -lh $SMOLBSD_IMAGE)"

          - name: Reset swtpm state (fresh PCR baseline)
            run: nu bin/swtpm-setup.nu --action reset --state-dir "$SWTPM_STATE"

          - name: Boot smolBSD with QEMU+swtpm (background)
            run: |
              nohup nu bin/qemu-smolbsd.nu \
                --image "$SMOLBSD_IMAGE" \
                --arch amd64 \
                --tpm \
                --tpm-state "$SWTPM_STATE" \
                --hostfwd-ssh "$SSH_PORT" \
                --mem 512M \
                --name smolbsd-tpm-ci \
                > /tmp/smolbsd-qemu-ci.log 2>&1 &
              echo "QEMU_PID=$!" >> "$GITHUB_ENV"
              echo "QEMU launched (PID $!)"

          - name: Wait for SSH boot-gate (120s timeout for KVM amd64)
            run: |
              for i in $(seq 1 120); do
                if sshpass -p "$SSH_PASS" ssh \
                     -o StrictHostKeyChecking=no \
                     -o UserKnownHostsFile=/dev/null \
                     -o ConnectTimeout=3 \
                     -p "$SSH_PORT" root@127.0.0.1 "echo UP" 2>/dev/null; then
                  echo "SSH_UP at ${i}s"
                  exit 0
                fi
                sleep 1
              done
              echo "ERROR: SSH did not respond within 120s"
              cat /tmp/smolbsd-qemu-ci.log | tail -30 || true
              exit 1

          - name: Run T1-T6 TPM acceptance suite
            run: |
              nu tests/bhyve-tpm-pcr-verify.nu \
                --host 127.0.0.1 \
                --port "$SSH_PORT" \
                --password "$SSH_PASS" \
                2>&1 | tee /tmp/smolbsd-t1t6-ci.toml

          - name: Report T1-T6 results in job summary
            if: always()
            run: |
              echo "## TPM T1-T6 Results" >> "$GITHUB_STEP_SUMMARY"
              echo '```toml' >> "$GITHUB_STEP_SUMMARY"
              cat /tmp/smolbsd-t1t6-ci.toml >> "$GITHUB_STEP_SUMMARY" || echo "(no results file)" >> "$GITHUB_STEP_SUMMARY"
              echo '```' >> "$GITHUB_STEP_SUMMARY"

          - name: Cleanup — shutdown guest and stop swtpm
            if: always()
            run: |
              sshpass -p "$SSH_PASS" ssh \
                -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null \
                -p "$SSH_PORT" root@127.0.0.1 "shutdown -p now" 2>/dev/null || true
              sleep 5
              nu bin/swtpm-setup.nu --action stop --state-dir "$SWTPM_STATE" || true
              kill "$QEMU_PID" 2>/dev/null || true
    ```
